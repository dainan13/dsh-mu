  let setOpen = false;
  function openSettings() {
    setLayer.style.cssText = 'position:fixed;inset:0;z-index:2147483500;display:grid;place-items:center;background:rgba(8,10,14,.32)';
    setLayer.innerHTML = TPL_SETTINGS;
    const dlg = setLayer.querySelector('.VOzbGW_panel');
    if (dlg) { dlg.style.width = 'min(980px, 92vw)'; dlg.style.height = 'min(700px, 86vh)'; }
    const close = setLayer.querySelector('.VOzbGW_close');
    if (close) {
      close.innerHTML = '<svg width="16" height="16" viewBox="0 0 16 16" fill="none" stroke="currentColor" stroke-width="1.3" aria-hidden="true"><path d="M4 4l8 8M12 4l-8 8"/></svg>';
      close.addEventListener('click', closeSettings);
    }
    setLayer.addEventListener('click', (e) => { if (e.target === setLayer) closeSettings(); });
    paintPane('models');
    setLayer.querySelectorAll('.VOzbGW_navCell').forEach(b => {
      b.addEventListener('click', () => {
        setLayer.querySelectorAll('.VOzbGW_navCell').forEach(x => x.classList.remove('VOzbGW_active'));
        b.classList.add('VOzbGW_active');
        paintPane(/模型/.test(b.textContent) ? 'models' : 'other');
      });
    });
    setOpen = true;
  }
  function closeSettings() { setLayer.innerHTML = ''; setLayer.style.cssText = ''; setOpen = false; }

  function paintPane(which) {
    const opts = setLayer.querySelector('.VOzbGW_options');
    if (!opts) return;
    opts.innerHTML = '<div id="muPane" style="height:100%"></div>';
    const host = opts.querySelector('#muPane');
    const sh = host.attachShadow({ mode: 'open' });
    sh.innerHTML = `<style>${CSS}</style>` + (which === 'models' ? MU_HTML : '<div class="mu"><h2>这一页没做</h2><p class="lead">原型里只做了「模型（mu）」右栏。</p></div>');
    if (which === 'models') wireMu(sh);
  }

  function wireMu(sh) {
    const form = sh.getElementById('addForm');
    sh.getElementById('addBtn').addEventListener('click', () => { form.hidden = false; sh.getElementById('addBtn').hidden = true; });
    sh.getElementById('cancelAdd').addEventListener('click', () => { form.hidden = true; sh.getElementById('addBtn').hidden = false; });
    sh.getElementById('saveAdd').addEventListener('click', () => { form.hidden = true; sh.getElementById('addBtn').hidden = false; toastish('已添加（原型）'); });
    sh.getElementById('delKey').addEventListener('click', () => toastish('删除要二次确认（原型）'));
  }
